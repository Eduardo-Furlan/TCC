    \centering
    \caption{Representação visual de uma rotação em \(\mathbb{R}^3\) em torno de uma reta \(\mathcal{L}\)}
    \label{fig:rotacao_r3}

    \begin{tikzpicture}[
            scale=1,
            transform shape,
            font=\sffamily,
            >=stealth,
            x={(-0.866cm,-0.5cm)}, y={(0.866cm,-0.5cm)}, z={(0cm,1cm)}
        ]

        \usetikzlibrary{calc, arrows.meta}

        \def\ColorAxis{\ifbool{darkmode}{gray!50}{black}}
        \def\ColorCubeA{\ifbool{darkmode}{DarkModeLink}{LightModeLink}}
        \def\ColorCubeB{\ifbool{darkmode}{DarkModeRed}{LightModeRed}}
        \def\ColorVector{\ifbool{darkmode}{DarkModeViolet}{LightModeViolet}}
        \def\ColorLine{\ifbool{darkmode}{teal}{teal}}

        \draw[->, \ColorAxis, thick] (0,0,0) -- (5.0,0,0) node[anchor=north east]{\(x\)};
        \draw[->, \ColorAxis, thick] (0,0,0) -- (0,5.5,0) node[anchor=north west]{\(y\)};
        \draw[->, \ColorAxis, thick] (0,0,0) -- (0,0,5.0) node[anchor=south]{\(z\)};

        \def\xa{1.5}
        \def\ya{2.0}
        \def\R{2.0}
        \def\zc{1.5}
        \def\cubeside{0.5}
        \def\thetaAngle{95}

        \pgfmathsetmacro{\Rprime}{sqrt((\R+\cubeside)^2+(\cubeside)^2)}
        \pgfmathsetmacro{\alphaOff}{atan2(\cubeside,\R+\cubeside)}

        \draw[ultra thick, \ColorLine] (\xa,\ya,-0.5) -- (\xa,\ya,4.5) node[above right]{\(\mathcal{L}\)};
        \node[circle, fill=\ColorLine, inner sep=1.5pt] at (\xa,\ya,\zc) {};

        \foreach \name/\x/\y/\z in {A/-1/-1/-1, B/1/-1/-1, C/1/1/-1, D/-1/1/-1, E/-1/-1/1, F/1/-1/1, G/1/1/1, H/-1/1/1} {
                \coordinate (\name 1) at ({\xa + (\R + \x*\cubeside)*cos(0) - \y*\cubeside*sin(0)}, {\ya + (\R + \x*\cubeside)*sin(0) + \y*\cubeside*cos(0)}, {\zc + \z*\cubeside});
                \coordinate (\name 2) at ({\xa + (\R + \x*\cubeside)*cos(\thetaAngle) - \y*\cubeside*sin(\thetaAngle)}, {\ya + (\R + \x*\cubeside)*sin(\thetaAngle) + \y*\cubeside*cos(\thetaAngle)}, {\zc + \z*\cubeside});
            }

        \draw[thick, \ColorCubeA, dashed] (A1) -- (B1) -- (C1) -- (D1) -- cycle;
        \draw[thick, \ColorCubeA, dashed] (A1) -- (E1);
        \draw[thick, \ColorCubeA, dashed] (B1) -- (F1);
        \draw[thick, \ColorCubeA, dashed] (C1) -- (G1);
        \draw[thick, \ColorCubeA, dashed] (D1) -- (H1);
        \draw[thick, \ColorCubeA, fill=\ColorCubeA, fill opacity=0.15] (E1) -- (F1) -- (G1) -- (H1) -- cycle;
        \draw[thick, \ColorCubeA, fill=\ColorCubeA, fill opacity=0.15] (B1) -- (C1) -- (G1) -- (F1) -- cycle;
        \draw[thick, \ColorCubeA, fill=\ColorCubeA, fill opacity=0.15] (C1) -- (D1) -- (H1) -- (G1) -- cycle;

        \draw[thick, \ColorCubeB, dashed] (A2) -- (B2) -- (C2) -- (D2) -- cycle;
        \draw[thick, \ColorCubeB, dashed] (A2) -- (E2);
        \draw[thick, \ColorCubeB, dashed] (B2) -- (F2);
        \draw[thick, \ColorCubeB, dashed] (C2) -- (G2);
        \draw[thick, \ColorCubeB, dashed] (D2) -- (H2);
        \draw[thick, \ColorCubeB, fill=\ColorCubeB, fill opacity=0.2] (E2) -- (F2) -- (G2) -- (H2) -- cycle;
        \draw[thick, \ColorCubeB, fill=\ColorCubeB, fill opacity=0.2] (B2) -- (C2) -- (G2) -- (F2) -- cycle;
        \draw[thick, \ColorCubeB, fill=\ColorCubeB, fill opacity=0.2] (C2) -- (D2) -- (H2) -- (G2) -- cycle;

        \draw[dashed, \ColorAxis] (G1) -- ($(G1)+(0,0,0.4)$);
        \draw[dashed, \ColorAxis] (G2) -- ($(G2)+(0,0,0.4)$);

        \begin{scope}[shift={(\xa, \ya, \zc + \cubeside + 0.4)}]
            \draw[thick, \ColorLine, dashed] (0,0,0) -- (\alphaOff:\Rprime);
            \draw[thick, \ColorLine, dashed] (0,0,0) -- ({\thetaAngle+\alphaOff}:\Rprime);
            \draw[->, >={Latex[length=2.5mm]}, thick, \ColorVector]
            (\alphaOff:\Rprime) arc[start angle=\alphaOff, end angle={\thetaAngle+\alphaOff}, radius=\Rprime]
            node[midway, above] {\(\theta\)};
        \end{scope}

    \end{tikzpicture}
