    \caption{Transformação Não-Isométrica}
    \label{fig:transformacao_nao_isometrica}

    \begin{tikzpicture}[
            scale=1,
            transform shape,
            font=\sffamily,
            >=stealth
        ]

        \def\ColorA{\ifbool{darkmode}{DarkModeLink}{LightModeLink}}
        \def\ColorC{\ifbool{darkmode}{DarkModeRed}{LightModeRed}}
        \def\ColorText{\ifbool{darkmode}{DarkModeText}{black}}

        \def\blobShape{plot [smooth cycle, tension=0.7] coordinates {(-1.5, -1) (0, -1.5) (1.5, -0.5) (1.2, 1.5) (-1, 1.2)}}

        \begin{scope}[shift={(0,0)}]
            \fill[\ColorA, opacity=0.15] \blobShape;
            \draw[thick, \ColorA] \blobShape;
            \node[\ColorA, font=\Large\bfseries] at (-1.5, 1.8) {\(X\)};

            \coordinate (x1) at (-0.5, 0.5);
            \coordinate (x2) at (0.5, -0.8);

            \node[circle, fill=\ColorText, inner sep=1.5pt, label={[text=\ColorText]left:\(x_1\)}] (nx1) at (x1) {};
            \node[circle, fill=\ColorText, inner sep=1.5pt, label={[text=\ColorText]below:\(x_2\)}] (nx2) at (x2) {};

            \draw[<->, \ColorText, dashed, thick] (nx1) -- (nx2) node[midway, left=2pt] {\(d_1\)};
        \end{scope}

        \begin{scope}[shift={(7, 0)}, scale=1.6]
            \fill[\ColorC, opacity=0.15] \blobShape;
            \draw[thick, \ColorC] \blobShape;

            \node[\ColorC, font=\normalsize\bfseries] at (-1.5, 1.8) {\(Z\)};

            \coordinate (z1) at (-0.5, 0.5);
            \coordinate (z2) at (0.5, -0.8);

            \node[circle, fill=\ColorText, inner sep=1pt, label={[text=\ColorText, scale=0.625]right:\(g(x_1)\)}] (nz1) at (z1) {};
            \node[circle, fill=\ColorText, inner sep=1pt, label={[text=\ColorText, scale=0.625]below:\(g(x_2)\)}] (nz2) at (z2) {};

            \draw[<->, \ColorC, dashed, thick] (nz1) -- (nz2) node[midway, left=2pt, scale=0.625] {\(d_2\)};
        \end{scope}

        \draw[->, \ColorText, thick, dotted] (nx1) to[out=20, in=170] (nz1);
        \draw[->, \ColorText, thick, dotted] (nx2) to[out=-20, in=190] (nz2);
        \node[\ColorText, font=\large] at (3.5, 1.5) {\(g\)};

        \node[\ColorText, align=center] at (3.5, -3) {
            \(d(x_1, x_2) \neq d(g(x_1), g(x_2))\)
        };

    \end{tikzpicture}
