% ── Referências bibliográficas (biblatex-abnt + biber) ───────
%
% Equivalências de comandos:
%   abntex2cite          →  biblatex-abnt
%   \cite{key}           →  \parencite{key}   (Autor, ano)
%   \citeonline{key}     →  \textcite{key}    Autor (ano)   ← também disponível como alias
%   \citeyear{key}       →  \citeyear{key}    (igual)
%   \bibliographystyle   →  não usado (o estilo fica em \usepackage[style=...]{biblatex})
%   \bibliography{file}  →  \addbibresource{file.bib}
%   \printbibliography   →  novo comando de impressão
%
% Campos do .bib:
%   Urlaccessdate = {DD de mês de AAAA}  ainda funciona com biblatex-abnt.
%   Para biblatex puro use urldate = {AAAA-MM-DD}.
%
\usepackage[%
    backend    = biber,
    style      = abnt,          % estilo autor-data ABNT (biblatex-abnt)
    backref    = false,          % "Citado na(s) página(s): X" na bibliografia
    backrefstyle = three,
    doi        = true,
    url        = true,
    isbn       = false,
    maxcitenames  = 3,
    maxbibnames   = 3,
]{biblatex}

% ─────────────────────────────────────────────────────────────
% COMANDO \linkcite — citação inline com hiperlink e ano
%
% Uso:
%   \linkcite{Cantor}{cantor1891}
%   → (Cantor, 1891) com link para a bibliografia
%
%   \linkcite[p.~42]{Cantor}{cantor1891}
%   → (Cantor, 1891, p. 42)
%
%   \linkcite{Autor1}{chave1}{Autor2}{chave2}
%   → (Autor1, ano1; Autor2, ano2)
%
% Com biblatex, \parencite{key} e \textcite{key} são os
% comandos canônicos. O \linkcite é mantido para
% compatibilidade com textos já escritos.
% ─────────────────────────────────────────────────────────────
\makeatletter

\def\linkcite{\@ifnextchar[{\@linkciteopt}{\@linkcitenormal}}

\def\@linkciteopt[#1]#2#3{%
    (%
    \hyperlink{cite.#3}{#2},%
    ~\hyperlink{cite.#3}{\citeyear{#3}}%
    \@linkciteloop{#1}%
}

\def\@linkcitenormal#1#2{%
    (%
    \hyperlink{cite.#2}{#1},%
    ~\hyperlink{cite.#2}{\citeyear{#2}}%
    \@linkciteloop{}%
}

\def\@linkciteloop#1{%
    \@ifnextchar\bgroup{%
        \@linkciteprocess{#1}%
    }{%
        \@linkciteend{#1}%
    }%
}

\def\@linkciteprocess#1#2#3{%
    ;%
    ~\hyperlink{cite.#3}{#2},%
    ~\hyperlink{cite.#3}{\citeyear{#3}}%
    \@linkciteloop{#1}%
}

\def\@linkciteend#1{%
    \ifblank{#1}{}{, #1}%
    )%
}

\makeatother
